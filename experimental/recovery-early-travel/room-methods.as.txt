      /** Only a negotiated shared destination owns this loading exception. */
      private function earlySupported():Boolean {
         if(!connected() || game.mapGeneration==null || !game.mapGeneration.earlyTravelSupported)return false;
         if(!host())return session.hostEarlyTravel==1;
         for each(var p:Object in session.peers)if(int(p.earlyTravel)!=1)return false;
         return session.peers.length>0;
      }
      private function offerEarly(info:Object):void {
         if(!earlySupported() || pending>0 || early.active || info.sceneMerged!==true || game.world==null)return;
         var g:Object=game.world.grafon;
         if(g==null || g.scenePreload!==true && g.sceneCullRender!==true)return;
         info.earlyId=earlyNonce+":"+(++earlySerial);info.earlyEpoch=epoch;
         if(!game.mapGeneration.acquireEarly(String(info.earlyId)) || !early.arm(info,game.world,epoch)) {
            game.mapGeneration.releaseEarly(String(info.earlyId));delete info.earlyId;delete info.earlyEpoch;
         }
      }
      private function armEarly(info:Object):void {
         if(early.active || info==null || String(info.earlyId || "")=="")return;
         if(!earlySupported() || pending>0){send({op:"earlyCancel",earlyId:String(info.earlyId)});return;}
         if(!game.mapGeneration.acquireEarly(String(info.earlyId))) {
            send({op:"earlyCancel",earlyId:String(info.earlyId)});return;
         }
         if(!early.arm(info,game.world,epoch)) {
            game.mapGeneration.releaseEarly(String(info.earlyId));throw new Error("Early travel prepare identity differs");
         }
      }
      public function get earlyTravelPending():Boolean{return early.active;}
      /** Called after Game has finished enterLand, spawn, effects and triggers. */
      public function nativeConstructed(nativeGame:Object):void {
         if(!early.active)return;
         try {
            game.refreshWorld();
            if(game.world!=null && game.world.game!==nativeGame)return;
            if(game.world==null || game.world.game!==nativeGame || !early.finishConstruction(game.world)) {
               if(!early.constructed){cancelEarly(true,true);return;}
               throw new Error("Early travel native completion changed");
            }
            if(!earlyControlHeld) {
               hold();earlyControlHeld=true;game.gg.controlOff();
               Log.d("RConnectEarly: constructed "+early.id+" room="+game.roomKey()+" t="+getTimer());
            }
            checkEarly();
         }catch(error:Error){fail(error);}
      }
      /** World calls this only through the exact Game hook. No normal-frame
       * refreshWorld or allocation when there is no active early transaction. */
      public function nativeLoadingHold(nativeGame:Object):Object {
         if(!early.active || !early.constructed)return null;
         if(game.world==null || game.world.game!==nativeGame || !early.valid(game.world) || game.isPlayerDead()) {
            cancelEarly(false,true);return null;
         }
         return early.ticket;
      }
      public function cancelNativeLoading(nativeGame:Object):void {
         if(early.active)cancelEarly(false,true);
      }
      private function checkEarly():void {
         if(!early.constructed)return;
         if(!early.valid(game.world) || game.isPlayerDead()) {cancelEarly(false,true);return;}
         if(blocked)return;
         if(host()) {
            if(nativeLand!==game.world.land) {
               nativeLand=game.world.land;nativeRoom=game.loc;
               epoch++;game.roomEpoch=epoch;records={};settledRoom=null;settledKey="";ready=false;
            }
            if(pending==0)begin();
         }else if(earlySeal!=null && !readySent) {
            var message:Object=earlySeal;earlySeal=null;
            if(game.roomKey()!=String(early.anchor.roomKey))throw new Error("Early destination room differs");
            nativeLand=game.world.land;nativeRoom=game.loc;hostInfo=message.info;
            pending=int(message.barrier);epoch=int(message.epoch);game.roomEpoch=epoch;
            ready=false;settledRoom=null;settledKey="";readySent=false;preparing=null;
            sendReady();
         }
      }
      private function finishEarly():void {
         if(!early.active || !early.constructed || blocked || !early.paintAllowed)return;
         if(!early.valid(game.world) || game.isPlayerDead()) {cancelEarly(false,true);return;}
         if(!early.viewReady(game.world,nativeFrame))return;
         if(!earlyViewFinished){game.finishFirstRaster();earlyViewFinished=true;}
         if(!host()) {
            if(!early.ackSent) {
               early.ackSent=true;send({op:"ack",barrier:pending,epoch:epoch,travelId:nativeTravel.id,earlyId:early.id});
               Log.d("RConnectEarly: ready "+early.id+" t="+getTimer());
            }
         }else if(early.peerReady) {
            send({op:"activate",barrier:pending,epoch:epoch,travelId:nativeTravel.id,earlyId:early.id});
            completeEarly();
         }
      }
      private function completeEarly():void {
         if(!early.valid(game.world) || !early.localReady || !earlyViewFinished || pending!=early.barrier || epoch!=early.epoch)
            throw new Error("Early activation is not ready");
         var id:String=early.id;
         if(!game.mapGeneration.releaseEarly(id))throw new Error("External travel hold did not release");
         if(nativeTravel.current!=null)nativeTravel.finish(nativeTravel.id);
         firstRaster.cancel();seedRaster.clear();seedDeadline=0;clearNativeEntry();nativeDeadline=0;
         game.activeTravelId="";preparing=null;pending=0;readySent=false;
         // The matching native player lost exactly one exit16 controlOn.
         if(earlyControlHeld) {game.gg.controlOn();game.world.pip.noAct=false;}
         earlyControlHeld=false;early.cancel();earlySeal=null;earlyViewFinished=false;
         release();Log.d("RConnectEarly: activated "+id+" t="+getTimer());
      }
      private function cancelEarly(notify:Boolean,discard:Boolean,disconnect:Boolean=false):void {
         if(!early.active)return;
         var id:String=early.id,constructed:Boolean=early.constructed;
         var same:Boolean=early.sameWorld(game.world);
         var resumable:Boolean=same && int(game.world.t_die)<=0 && int(game.world.comLoad)<0
            && int(game.world.ng_wait)<=0 && !game.isPlayerDead();
         if(notify && connected())send({op:"earlyCancel",earlyId:id});
         if(early.paintAllowed && resumable && connected() && !disconnect) {
            // Keep ownership until disconnect: even a fully restored but not
            // activated world must not be released by a malformed old packet.
            blocked=true;hold();status="房间同步失败，请断开连接后重试";return;
         }
         if(!game.mapGeneration.releaseEarly(id)) {
            blocked=true;hold();status="房间同步失败，请断开连接后重试";return;
         }
         early.cancel();earlySeal=null;earlyViewFinished=false;
         // A native load/death/replacement owns its own control and pause now.
         // Never transplant the old saved pause into a new Game in this World.
         if(!resumable && pauseWorld===game.world)pauseWorld=null;
         if(!resumable){ready=false;settledRoom=null;settledKey="";}
         if(resumable && earlyControlHeld && (disconnect || int(game.world.t_exit)<=16)) {game.gg.controlOn();game.world.pip.noAct=false;}
         release();
         earlyControlHeld=false;
         if(discard){pending=0;readySent=false;requested=false;nativeRoom=null;}
      }
